extends SceneTree

func _initialize() -> void:
	call_deferred("run")

func shot(file: String) -> void:
	await process_frame
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://tests/"+file+".png")

func run() -> void:
	var game=load("res://scenes/main.tscn").instantiate();root.add_child(game)
	await shot("menu")
	game.close_modal();await shot("vigil")
	game.interact(game.area.points[0]);await shot("mara")
	game.close_modal();game.enter_area("road");game.hero.position=Vector2(1140,1470);game.camera.position=game.hero.position
	await shot("road")
	game.enter_area("bannerfall");game.hero.position=Vector2(1000,820);game.camera.position=game.hero.position
	await shot("bannerfall")
	quit()
